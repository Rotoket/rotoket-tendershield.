## 🔧 Преимущества

### До:
```typescript
// Разные способы обработки ошибок
if (!resp.ok) {
    const text = await resp.text();
    throw new Error(`Ошибка: ${resp.status} ${text}`);
}
```

### После:
```typescript
// Единый подход с типизированными ошибками
if (!resp.ok) {
    const apiError = await parseAPIError(resp, 'Ошибка загрузки');
    throw new APIErrorException(apiError);
}
```
